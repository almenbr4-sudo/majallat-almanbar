يومكم آتٍ قريبا / علي عبد ربّ النّبـــي

• 2018-07-20

*********(يومكم آتٍ قريبا )*********** قــد سئمنا مـنْ حيــاةٍ فيـها للغــــدر مَســــارْ سـاســـةٌ دون حَيـــاءٍ ومســاعيــرُ كبــــــارْ قــاومـــوا كـلّ صـــﻻحٍ وجــرى منهــم دمـــارْ ســاندوا كــلّ لئيـــمٍ كــان للــمــال حـــــوارْ قـــدّمــوا دون حسـابٍ كــان للمــوت حصــار حــاصــرونا بســــﻻحٍ فيــه للشّــرِّ انتصـــارْ ســرقَ الخــــوف قلـوبا منــه لـــم ينجُ الصّغارْ ليلُهــم خــوفٌ ورعـبٌ مــنْ كــوابـيـس تُـثـارْ ونهــارٌ فيــه خطـــفٌ ليس للطّــفل اعتبــارْ يُقتَلُ الطّفــلُ انتقامــا حين يـشتـــدّ الحــوارْ إدفعــــوا المــــالَ وإﻻّ طِفلُكـــم ســوف يُضارْ بئــس مـنْ ضلّوا ضﻻﻻ فيــه مَقـــتٌ وخســارْ إعلمـــوا مهما ظلمتــم لـن يواتيكــــم فــرارْ يومـكــُم آتٍ قــريبـا وهـــو ﻻشــكّ المســارْ فـي لظــى نار السّعـير وهي مثــــوى وقـــرارْ علي عبد ربّ النّبـــي
