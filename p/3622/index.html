وركبت ُ صبـري / علي عبد رب النبي

• 2018-07-31

( وركبت ُ صبـري) حـُبٌّ وحُـزنٌ يَنخران كِيـانــي ومُـدى التّمنّـي قطّعتْ أركانـي ﻻ أستفيقُ وﻻ أرانـي واعيـا وإذا أفقــتُ بــدا الفــؤادُ يعانـي الحُبّ معْنًـى يستبيحُ مشاعري والحـزنُ يُقصـي فرحتي بـيماني حُـزني يُواري مطمعـي بوصالها ووِصـالهــا متُمَنــّعُ الأركــــاِنِ ياهـذه يكفي العـذابُ لمهجتي هـل تُنكـريـن تَعلّقـي وحنانـي هـيّا أزيحي السّـدّ حول عواطفي وتمتعــي بقصــائدي ودنانــــي إنـّي سأمضـي رغم حزنـي آمﻻ مَجــرى عطــاءٍ غامـرًا لجَنانــي وركبـتُ صبـري فـي جوامع عُـدّتي وأعـذتُ نفسي بالمُنـى الإنسانـي مادام لـي أمـلٌ يداعبُ مهجتـي لــن أستمـرّ مـع القنـوط ثوانــي علي عبدرب النبي
