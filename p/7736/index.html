ٱمنت بالله - علي عبدربّ النّبـــي - ليبيا

• 2019-07-17

*****************(ٱمنت بالله )************ آمنــت ُبالله العظيم مُسلّمــا أمــري لربــّي راضيــا مرضيا مالي سوى ربي يجيب ضراعتي فهــو الإله وقـد عبـدت عليا سبحـانــه قد مدّني بمواهبـي حمـدا وشكـرا واجبـــا مقضيا من فضله نلت المنى بعقيدتي لا شــيء غيـر الله أدعــو وليا من هديه كوني حنيفا مسلمـا وحـــّدت ربّــي واتبعــتُ نبيـا ياخالقــي أكرمتنـي فهديتنـي سبحــان ربّــي بكــرة وعشيا ياربُ ثبــّت في الفــؤاد محبة فيهــا أكــون مــوافيا وصفيّـا يامــن إليـه توجهي وتضرعي اجعـلْ فـؤادي خاشعـا وتقيّــا خلص فـؤادي من نفاق مكابر وابعده عنـّي ما دوامــي حيّـا ألقاك ربّي يوم حشري غافرا مـاكان منّــي فاضحـا ورديّــا يامــن وسعت مراحما لاتنتهي فيهـا نصيبـي لا تـذرنِ قصيا علي عبدربّ النّبـــي
