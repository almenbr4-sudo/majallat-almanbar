كُنْ على أَمَل - حامد حفيظ - ليبيا

• 2021-09-11

#كُنْ_على_أَمَل ﻛُـﻦْ ﻋَـﻠَﻰٰ يُمْنٍ ، ﻭَﺻَــﺪِّﻕْ ﺃَﻥَّ ﻣَـﺎ ﺗَـﺮﺟـُـــﻮ تَحـَــقَّـقْ وﺍﺣـﺴِـﻦِ ﺍﻟــﻈَــﻦَّ ﺑِـﺮﺑــﻲ ﻛَـﻄُـﻴُــﻮﺭِ ﺍﻟـﺠَــــﻮِّ ﺗُــﺮْﺯَﻕْ ﺃَﻧْﺖَ ﺗَـهـــذِي ﻛُــﻞَّ ﻳـَـــﻮﻡٍ وَالـقَـطـَـا غَنَّىٰ وَزَﻗـْــــﺰَﻕْ أنْتَ تَـجـْـثُـو مُـسـتكـينًـا وَحَــمَـامُ الـحَـقـلِ حَــلَّـقْ ﻭَالفَـرَﺍﺵُ انسَــاقَ ﺃَوهَـىٰ ﻣِـﻨْـﻚَ ، ﻟَﻜِـﻦْ ﻟَـﻴﺲَ ﻳَﻘْــﻠَﻖْ سـُـــلَّـمٌ الـمَـجـدِ طـَـويـْـل كـُــنْ صـَـبـُـورًا وَتَـسـَــلَّـقْ يَبـْــــــدَأ الــورد نـتـــــوءًا بـَـعـــــد حـِــينٍ يَـتـفَــتَّـقْ وَغـَـمـَـامُ الـغــيــم قَـطـْـرٌ بـعــْــدَ رَهـْــنٍ يَـتــَــدَفَّـق ﻛُـﻦْ ﻣِــﻦَ ﺍﻷَﺣــﻼﻡِ ﺃَﺑـﻬَـﻰٰ ﻭ ﻣِـﻦَ ﺍﻟﻨَـﺴْـﻤَـﺎﺕِ ﺃﺭﻓـَــﻖْ ﻭَ ﻣِــﻦَ ﺍﻷَﻧــﻬَـــــﺎﺭِ ﺃﻧْـﻘَـﻰٰ ﻭَ ﻣِــﻦَ ﺍﻟـﻮﺩﻳَـﺎﻥِ ﺃﻋــﻤَــﻖْ ﺇِﻥْ ﺧَﺸِﻴﺖَ ﺍﻟﻤـﻮﺕَ ﻳَﻮﻣـﺎً ﻓِﻲ ﺍﻟﻤِﻴَﺎﻩِ، ﺳَﻮﻑَ ﺗَـﻐْـﺮَﻕْ ﺇِﻥْ أَﺿَـﻌـْﺖَ ﺍﻟـﻌُـﻤْـﺮَ هـَدْرَاً ﻓِﻲ ﺍﻟﻬُﻤﻮﻡِ ﺳَﻮﻑَ ﻳُﺴـﺮَﻕْ ﺇِﻥْ ﺷَﻜُـﻮﺕَ ﺍﻟﻔَـﻘْﺮَ ﺿِـﻴﻘـﺎً فَـاتسـَاع الـعيش ﺃَﺿـْﻴـَﻖْ ﻻ ﺗَـﻘُـــﻞْ ﺑـﺎﻟــﻮَﺭﺩِ ﺷُـــﻮﻙٌ ﺑَـﻞْ رَحِـــيـقٌ ، ﻭ ﺗَـﻨـﺸَّــﻖْ ﻻ ﺗَـﻘُـﻞْ ﻟِـﻠـﻨَـﺤـــﻞِ ﻭَﺧـْــﺰٌ ﺑَـﻞْ ﻟَــﻪُ شَــهْــدٌ ﻣُـﺨَـــﻠَّـﻖْ ﻻ تَـــلُـذْ بـِالـيَـأسِ هَـوْنَـاً ﺫَﺭْﻩُ ﺑَـﺎﺏَ ﺍﻟـﻴَـﺄﺱِ ﻣُـﻐـﻠَـﻖْ ﺇِﻥْ ﺃَﺻَـﺎﺏَ ﺍﻟـﻘَـﻠْﺐَ نَـــزْغٌ ﻓَـﺎﺳـﺘَـﻌِـﻦْ ﺑِـﺎﻟﻠﻪِ تُـعــتَـقْ حامد حفيظ - ليبيا
