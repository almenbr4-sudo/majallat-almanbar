البﻻغ / علي عبد رب النبي. الزاوية. ليبيا

• 2018-07-05

******************( البﻻغ )******** جئت الدنا يوما وقــلبي يخفق رمتُ الحياة سعادة تتـــألقُ أعطِيتُ قلبا ﻻ يقـــرّ قراره حتى يرى مجد العروبة يشرقُ قومي أنادي فاسمعوا وتوثبوا صهيون يسعى بالفناء يُحَرِّقُ أهداه بلفــرْ أرضنا وسماءنا أرجاه فينا ذلـّــةٌ وتفــرّقُ أغــراه غربٌ حاقدٌ متعصِّــبٌ فاشتدّ يفنى جمعَنا وُيمزّقُ أطــغاه سـامٌ بالسﻻح يمـــدّهُ أجـــراه طوفانا يُميت ويُغــرِقُ أطـــراه فعﻻ في مجالس شرِّه قتل العروبة واجبٌ ﻻيُسـبَُقُ أخفــاه جرما ﻻ يُقام خفــاؤهُ صُنع الفناء جريمة تتحققُ أعفــاه من صلب المسيح وقتله ناداه إنّــي قد رأيتك تصدقُ كفرت عن صلب المسيح وقتله إرهــابك للعْرب فيه تألّـــقُ صهيونُ والصلبانُ قلـبٌ واحدٌ إذ ليس لﻹســﻻم منهــم موثقْ ﻻينــزل الربّ المسيحُ بأرضهم حتى نرى اﻹسﻻمَ منها يُزهقُ شدوا الخناق على العراق نكاية مدوا على ليبيا حصارا يرهقُ هذي الجزائر فاضربوها ببعضها أرموا على اﻹسﻻم جرما يُلصَقُ أرض الكنانة للعروبة موطنٌ فتدبروا كيف نراها تُمحَقُ أرضاهــمو مــوتٌ لــنا بتآمـــرٍ ماكــان لﻹســﻻم فيها خنــدقُ ديـــن اﻹلــه مـبرأ من فريــةٍ كــان الصليبُ يعـدّهـــا ويوثق مالي أرى اﻷفغان يفني بعضه بالله هــل هــذي فعال تصدقُ مــات الجميعُ وﻻ ثواب مجاهدٍ ضــلوا سبيل الله حين تفرقوا ســروا عدوا قد أباح ديارهـــم صــبوا على اﻹخــوان نارا تحرقُ أي الجــهاد فــعلهم بأقـــاربٍ بئس الذي فعلوا وبئس المأزقُ بعـــض مثالٍ للفنــاء ضربتــه ألــف وألــف غيرها تتعمـّـقُ مــاذا دهــانا في صميــم قلوبنا نــرجــوا ودادا من عــدو يحنق ُ مــن قــال أن الغرب يصـفي ودّه للعُــرب فــهو جاهـل بل أخرقُ خاب الذي يسعى إليهم صاغرا عبـدا ذلـــيﻻ ﻻتــراه ينطـــقُ يرضــى الخيــانة ﻻيراها سبةً يسعــى إليـها طائــعا يتحـّـرّقُ باعوا الضمير مع الحياء كليهما ﻻخير يرجى من لئيم يمـــرقُ ﻻشيء قبل حسابهم وعقــابهم لسنا نرى شيئا ســـواه يسبـــقُ حتى نطهر جمعنا من خــائـن نهفوا إلى الهيجاء جمعا يصدقُ ياأمتــي إنــي أراك حليمــــةً لما نجا من سيفك المتفيهــقُ الحلم ضعف حين ينجي خائنا جر الدمار وانثنـــى يتشــدقُ إنــي أمين واﻷمانـة منهـجــي حبــي لقومـي صـادق متعمـقُ ياأمتــــي إنـــي أراك بعيــدةً عن وحدة فيها العروبة تنطقُ قلبــا أبيــا ﻻيقــــر قــراره حتى يراها غايـــة تتحقــقُ ﻻتسلكي غير الطريق لوحدة إن شئت دفعا للكوارث تحــُدقُ إن شئت عـزا فارفعـي راياتــه حتــى تـردي طامعا ﻻيـرفـقُ ياأمتــي ﻻتيأسي مــن حاضــر فيــه العــدو قــادر متفـــوقُ إن تصدقي قوﻻ وفـعﻻ فاعلمـي أن العــدو عاجــز بــل مخفــقُ إن تنصــري الله فأنــت بقــوة مدعومــة مــن ربهــا تتدفــقُ هـذا كتــاب الله منه فانهـلــي ماخــاب من أضحـى به يتعلـقُ ياأمتــي ﻻتقنطــي ﻻتيأســي إني أراك عزيمــة ﻻتــُرهــقُ أحي من التاريخ مجدا شامخــا فيه علوت وليس غيرك ينطقُ كوني الوريث والمثال كليهمــا من غيــرك ياأمتــي يتفــوقُ هــذا بـــﻻغ صــادق ياأمتــي يا أمـّـة اﻹســﻻم هـذا موثـقُ بقلم : علي عبد رب النبي. الزاوية. ليبيا اﻷحد. 20. 11. 1994
