# مقاطع_من_معارضة_شعرية يَا لَيْلُ الشَّعْبُ مَتَى غَدُهُ . . . . أَقِيَامُ السَّاعَةِ مَوْعِدُهُ يَا وَطَنَاً تَاهَ بِلَا رَشَدٍ . . . . لَمَّا ضَيَّعَه مُرْشِدُهُ نَارٌ لِلْحَرْبِ تُمَزِّقُهُ . . . . و تُدَمِّرُهُ و تُشَرِّدُهُ و مَجَالِسُهُ و حَكَائِمُهُ . . . . كُثْرٌ وَ تَعَقَّدَ مَشْهَدُهُ خُوِّنَ أَوْ غُيِّبَ مُصْلِحُهُ . . . . و تَوَلَّى أَمْرَهُ مُفْسِدُهُ مَرَّتْ خَمْسُونَ و لَا زِلْنَا . . . . نَبْحَثُ عَن وَطَنٍ نَفْقِدُهُ أَيَكُونُ شِعَارًا أَوْ عَلَمَاً . . . . و نَشِيْدَاً عَنْهُ نُرَدِّدُهُ أَيْنَ الْمُخْتَارُ و إِخْوَتُهُ . . . . و الْمَلِكُ(ادْرِيْسُ)مُوَحِّدُهُ مَجْدٌ اِنْ خَلَفٌ ضَيَّعَهُ . . . . يَبْقَى التَّارِيخُ يُخَلِّدُهُ لَعَنَاتُ الْلَّهِ تُحيْقُ بِمَنْ . . . . بِالْغَدْرِ لَهُ امْتَدَّتْ يَدُهُ يَا وَطَنَاً كَانَتْ مِحْنَتُهُ . . . . فِي نِعَمٍ ظَنَّ سَتُسْعِدُهُ مَحْرُومٌ مِنْهَا مُذْ وُجِدَتْ . . . . و النَّاسُ عَلَيْهَا تَحْسُدُه أيَضِيْقُ و تِلْكَ رَحَابَتُهُ . . . . أَيِجُوْعُ و تِلْكَ مَوَارِدُهُ و الذَّهَبُ الْأَسْوَدُ هَلْ قَدَرٌ . . . . (لَا يُعْرَفُ إِلَّا أسْوَدُهُ) يَا شَعْبَاً حِيْنَ تَقَلَّدَهُ . . . . أَسْوَءُ مَنْ فِيهِ و أَحْقَدُهُ لَمْ يُنْكِرْهُ بَلْ صَيَّرَهُ . . . . عَنْ جَهْلٍ صَنَمَاً يَعْبُدُهُ لِيُذَلَّ و تُخْطَفَ أُمَّتُهُ . . . . مِنْ طَاغٍ, مَنْ ذَا مُنْجِدُهُ اِنْ يَأْبَى الْعَبْدُ تَحَرُرَهُ . . . . كَيْفَ سَيُعْتِقُهُ سَيِّدُهُ و أَخِيرًا أَطْلَقَ مَارِدَهُ . . . . طُوْفَانُ الْمَوْتِ تَمَرُّدُهُ
