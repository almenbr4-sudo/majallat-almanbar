كسير الجناح - حامد حفيظ - ليبيا - صحيفة المنبر

• 2019-01-23

ﻛُﻨْﺖُ ﺍﻟﺬﻱ ﺯﺭﻉَ ﺍﻟﻘـﻠﻮﺏَ ﺳَـﻤﺎﺣـﺔً ﻓَﺠَﻨﻴﺖُ جُرْحاً ﻣِﻦْ ﺟَﻤِﻴﻞِ ﺳَﻤَﺎﺣﻲ ﻛَﺎﻟـﻄﻴﺮ ﻓﻲ جوِّ ﺍﻟﺴـﻤﺎﺀِ ﻣُﺤَـﻠِّﻘَـﺎً ﺭﻭﺣــــﺎً ﺗَـﻄِـﻴـﺮُ ﻛَــﺴَـﺎﺋِـﺮِ ﺍﻷَﺭﻭﺍﺡِ ﻭﺍﻟﻴـﻮﻡَ غافَلَنِي ﺍﻟـزمانُ ﺑِﺨِﻨْﺠَﺮٍ ﻣَــﺴـﻤُـﻮﻣـﺔٍ ﻓَـﺘَـﻴَﺘّـﻤَـﺖ ﺃَﻓـْـﺮَﺍﺣِﻲ ﻫَﺪَﻣَﺖ ﺭﻳﺎﺡُ ﺍﻟﺨﻮﻑِ ﻋُﺶَّ ﺳﻌﺎﺩَﺗِﻲ ﻇُـﻠﻤـﺎً، ﻭﺑَـﺮﺩُ ﺍليأسِ ﺷَـﻞَّ ﺟَﻨَـﺎﺣﻲ ﻭﺭﺳَـﻤـﺖُ ﺃَﺣـﻼﻣِﻲ بِـحِـبـرِ ﺑَـﺮﺍﺋَﺘِﻲ ﻟَـﻜِـــﻦَّ ﺯَﻣَـﻨِــﻲ ﻟِـﻠـﺒـَــﺮَﺍﺀَﺓِ ﻣَـــﺎﺣِـﻲ ﺃَﺧْﻔَﻴﺖُ جُرحِي ﻋَـﻦْ ﺃُﻧَﺎﺳِﻲ ﻋَـﻨْـﻮَﺓً ﻭَﺃَﺧَــﺬﺕُ ﺃُﻇـﻬِـﺮُ ﺑَﺴﻤَﺘِﻲ ﻭﻣُـﺰَﺍﺣِﻲ ﻓَـﺄﻧَــﺎ ﺳَـﺠِـﻴﻦٌ ﺩُﻭﻥَ ﺣُـﻜْـﻢٍ ﺻــﺎﺩِﺭ ﻟَـﻢْ ﻳُﻄـﻠِـﻖْ ﺍﻟـﻬَـﻢَّ ﺍﻟﺠَـﺜِﻴﻢَ ﺳَـﺮَﺍﺣﻲ ﻭَﺃَﻧَـﺎ ﻋَـﻠَﻰ ﻣَـﻮﺝِ ﺍﻟﺤَـﻴـَﺎﺓِ ﺳَـﻔِـﻴﻨَـﺔ ﺃَﻧـﺴَـــﺎﻕُ ﺑَـﻴْﻦَ ﻋَــﻮَﺍﺻِـﻒٍ ﻭَﺭِﻳَـــﺎﺡِ ﻭَﺍﻟـﺠُـﺮﺡُ مِهرَاقٌ ﺇﺫﺍ داويتـــﻪُ يتمخَّض الأوجاعَ في أقداحي ﻭﺍﻟـيأس ﻳَـﺴْـبِقُـﻨِـﻲ ﺇﺫﺍ ﺟَــﺎﺭِﻳﺘُـــﻪُ ﻣُـﺘـﺴَـﺎﺭِﻋَـﺎً ﻓِﻲ ﻏَــﺪﻭَﺗِﻲ ﻭَﺭَﻭَﺍﺣِـﻲ ﻳَﺎ ﻟَـﻴﺘَـﻬَـﺎ ﻛَـﺎﻧَﺖ ﺟِــﺮَﺍﺣِﻲ ﻛِـﺬْﺑَﺘــﺔً ﺃَﻭ ﺑَـﻌـﺾ ﺃَﺿـﻐَﺎﺙٍ ﻟِﺄَﻏـﺪُﻭَ ﺻَـﺎﺣِﻲ ﻟَـﻜِـﻨَّــﻬَـﺎ ﻧَـﺰَﻓَـﺖْ ﻭَﺟُــﻦَّ ﺃَﺟِــﻴـﺠُـﻬَـﺎ ﻛَـﺎﻟـﻨَــــﺎﺭِ ﺣِــﻴﻦَ ﺗَـﺸـﺐُّ ﺑِـﺎﻷَﻟْـــﻮَﺍﺡِ ﻳَﺎ ﻣَﻦْ ﺇﺫَﺍ ﺳَـﺎﺋَﺖْ ﺣَـﻴَـﺎﺗِﻲ ﺗَﻔْـﺮَﺣُـﻮﺍ ﻳﺎ ﻣَﻦْ ﻳَﺴُـﺆْﻛُﻢ ﻣَـﻐْﻨَـﻤِﻲ ﻭَﻧَﺠَــﺎﺣِﻲ ﻓَﻠﺘَﺠـﻌَـﻠُﻮﺍ ﻗَـﻠْﺒِﻲ ﻣَـﻜَـﺎﻥَ ﻗُـﻠُﻮﺑَـﻜُـﻢ ﺣَـﺘَّﻰ ﺗَﻘـﻴِﺴُﻮﺍ ﺣـﺴﺮَﺗِﻲ ﻭﺟِـﺮَﺍﺣِﻲ __________________________ كسيرُ الجناح/ بقلم/ حامد حفيظ
